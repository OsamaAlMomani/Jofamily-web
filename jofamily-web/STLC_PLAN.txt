import { describe, it, expect, beforeEach } from 'vitest';
import { safeAddIceCandidate, flushPendingCandidates } from '../ice';

describe('ICE Candidate Buffering', () => {
  let mockPC;
  let pendingRef;
  const mockCandidate = {
    candidate: 'candidate:1 1 udp 2130706431 192.168.1.1 50000 typ host',
    sdpMid: '0',
    sdpMLineIndex: 0
  };

  beforeEach(() => {
    pendingRef = { current: [] };
    mockPC = {
      remoteDescription: null,
      addIceCandidate: vi.fn().mockResolvedValue(undefined)
    };
  });

  it('buffers candidates when no remote description', async () => {
    await safeAddIceCandidate(mockPC, mockCandidate, pendingRef);
    
    expect(pendingRef.current).toHaveLength(1);
    expect(pendingRef.current[0]).toEqual(mockCandidate);
    expect(mockPC.addIceCandidate).not.toHaveBeenCalled();
  });

  it('adds candidate directly when remote description exists', async () => {
    mockPC.remoteDescription = { type: 'offer', sdp: 'mock-sdp' };
    
    await safeAddIceCandidate(mockPC, mockCandidate, pendingRef);
    
    expect(pendingRef.current).toHaveLength(0);
    expect(mockPC.addIceCandidate).toHaveBeenCalledWith(mockCandidate);
  });

  it('flushes pending candidates after remote description set', async () => {
    // Buffer candidates
    await safeAddIceCandidate(mockPC, mockCandidate, pendingRef);
    await safeAddIceCandidate(mockPC, mockCandidate, pendingRef);
    expect(pendingRef.current).toHaveLength(2);
    
    // Set remote description and flush
    mockPC.remoteDescription = { type: 'offer', sdp: 'mock-sdp' };
    await flushPendingCandidates(mockPC, pendingRef);
    
    expect(pendingRef.current).toHaveLength(0);
    expect(mockPC.addIceCandidate).toHaveBeenCalledTimes(2);
  });
});

import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { initializeApp } from 'firebase/app';
import { getFirestore, connectFirestoreEmulator, deleteDoc, doc } from 'firebase/firestore';
import { createRoomDoc, writeRoomAnswer, listenForAnswer } from '../firestoreRooms';

describe('Firestore Room Operations (Integration)', () => {
  let db;
  const testRoomId = 'test-integration-room';

  beforeAll(() => {
    const app = initializeApp({ projectId: 'test-project' });
    db = getFirestore(app);
    connectFirestoreEmulator(db, 'localhost', 8080);
  });

  afterAll(async () => {
    // Cleanup
    await deleteDoc(doc(db, 'rooms', testRoomId));
  });

  it('creates room document with offer', async () => {
    const mockOffer = {
      type: 'offer' as const,
      sdp: 'mock-sdp-offer'
    };

    await createRoomDoc(db, testRoomId, 'Alice', mockOffer);
    
    const roomRef = doc(db, 'rooms', testRoomId);
    const snap = await getDoc(roomRef);
    
    expect(snap.exists()).toBe(true);
    expect(snap.data().createdBy).toBe('Alice');
    expect(snap.data().offer).toEqual(mockOffer);
  });

  it('writes answer to existing room', async () => {
    const mockAnswer = {
      type: 'answer' as const,
      sdp: 'mock-sdp-answer'
    };

    await writeRoomAnswer(db, testRoomId, 'Bob', mockAnswer);
    
    const roomRef = doc(db, 'rooms', testRoomId);
    const snap = await getDoc(roomRef);
    
    expect(snap.data().answeredBy).toBe('Bob');
    expect(snap.data().answer).toEqual(mockAnswer);
  });
});

test.describe('WebRTC Connection', () => {
  test.slow(); // Mark as slow test (can take 10-30 seconds)

  test('two users can establish connection', async ({ browser }) => {
    // Create two separate browser contexts (simulate 2 users)
    const contextA = await browser.newContext({
      permissions: ['camera', 'microphone']
    });
    const contextB = await browser.newContext({
      permissions: ['camera', 'microphone']
    });

    const pageA = await contextA.newPage();
    const pageB = await contextB.newPage();

    try {
      // User A creates room
      await pageA.goto('/rooms');
      await pageA.getByLabel('Name').fill('Alice');
      await pageA.getByRole('button', { name: /create room/i }).click();
      
      // Get the room URL
      const roomUrl = pageA.url();
      console.log('Room URL:', roomUrl);

      // User A starts camera
      await pageA.getByRole('button', { name: /start camera/i }).click();

      await pageB.goto(roomUrl);

7.2 SECURITY TESTING
---------------------

Test 1: Firestore Security Rules Validation
File: firestore.rules
----------------------
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Rooms collection
    match /rooms/{roomId} {
      // Anyone can read room documents
      allow read: if true;
      
      // Only authenticated users can create rooms
      allow create: if request.auth != null;
      
      // Only room creator can update/delete
      allow update, delete: if request.auth != null 
                              && request.auth.uid == resource.data.createdBy;
      
      // ICE candidates subcollections
      match /callerCandidates/{candidateId} {
        allow read, write: if true; // Public for signaling
      }
      match /calleeCandidates/{candidateId} {
        allow read, write: if true; // Public for signaling
      }
    }
  }
}

APPENDIX B: COMMON WEBRTC TESTING GOTCHAS
================================================================================

1. PERMISSIONS IN AUTOMATED TESTS
Problem: Browser permissions don't work the same in headless mode
Solution: Use context.grantPermissions() in Playwright

2. ICE CANDIDATE TIMING
Problem: Candidates may arrive before remote description is set
Solution: Use buffering mechanism (already implemented in ice.ts)

3. STUN SERVER RELIABILITY
Problem: Public STUN servers can be unreliable
Solution: Use multiple STUN servers; consider dedicated TURN server

4. MOBILE TESTING LIMITATIONS
Problem: Real devices behave differently than emulators
Solution: Always test on physical iOS and Android devices before production

5. CLEANUP IMPORTANCE
Problem: Forgotten cleanup causes memory leaks in long sessions
Solution: Always stop tracks and close peer connections on unmount

6. FIRESTORE EMULATOR DIFFERENCES
Problem: Emulator security rules may behave slightly differently
Solution: Test in real Firebase project before final production deploy
